2.7.3 · Ví dụ 3
Lọc thông thấp Gauss trong miền tần số
import cv2, numpy as np
img = cv2.imread("anh.jpg", cv2.IMREAD_GRAYSCALE).astype(np.float64)
M, N = img.shape
F = np.fft.fftshift(np.fft.fft2(img)) # DFT + dịch tâm
u = np.arange(M) - M//2
v = np.arange(N) - N//2
V, U = np.meshgrid(v, u)
D = np.sqrt(U**2 + V**2) # khoảng cách tới tâm
D0 = 40
H = np.exp(-(D**2) / (2 * D0**2)) # thông thấp Gauss
G = F * H # áp bộ lọc
ket_qua = np.real(np.fft.ifft2(np.fft.ifftshift(G)))
ket_qua = np.clip(ket_qua, 0, 255).astype(np.uint8)
